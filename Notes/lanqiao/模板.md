# 七、双指针模板

## 1. 最长不重复子数组

```python
from collections import defaultdict

a = [1, 2, 1, 3, 2, 4]
n = len(a)

cnt = defaultdict(int)
left = 0
ans = 0

for right in range(n):
    cnt[a[right]] += 1

    while cnt[a[right]] > 1:
        cnt[a[left]] -= 1
        left += 1

    ans = max(ans, right - left + 1)

print(ans)
```

# 十、BFS 模板

## 1. 图 BFS

```python
from collections import deque

n = 5
graph = [[] for _ in range(n + 1)]
dist = [-1] * (n + 1)

def bfs(start):
    q = deque()
    q.append(start)
    dist[start] = 0

    while q:
        u = q.popleft()

        for v in graph[u]:
            if dist[v] == -1:
                dist[v] = dist[u] + 1
                q.append(v)
```

## 2. 网格 BFS 最短路

```python
from collections import deque

n, m = 3, 3

grid = [
    ['S', '.', '.'],
    ['#', '#', '.'],
    ['.', '.', 'T']
]

dist = [[-1] * m for _ in range(n)]
dirs = [(1, 0), (-1, 0), (0, 1), (0, -1)]

def bfs(sx, sy):
```

# 十二、并查集模板

```python
class DSU:
    def __init__(self, n):
        self.parent = list(range(n + 1))
        self.size = [1] * (n + 1)

    def find(self, x):
        while self.parent[x] != x:
            self.parent[x] = self.parent[self.parent[x]]
            x = self.parent[x]
        return x

    def union(self, a, b):
        ra = self.find(a)
        rb = self.find(b)

        if ra == rb:
            return False

        if self.size[ra] < self.size[rb]:
            ra, rb = rb, ra

        self.parent[rb] = ra
        self.size[ra] += self.size[rb]
        return True

    def same(self, a, b):
        return self.find(a) == self.find(b)
```

使用：

```python
dsu = DSU(10)

dsu.union(1, 2)
dsu.union(2, 3)

print(dsu.same(1, 3))  # True
```

# 十三、最短路模板

## 1. Dijkstra

适合边权非负的最短路。

```python
import heapq

def dijkstra(n, graph, start):
    INF = 10 ** 18
    dist = [INF] * (n + 1)
    dist[start] = 0

    heap = [(0, start)]

    while heap:
        d, u = heapq.heappop(heap)

        if d != dist[u]:
            continue

        for v, w in graph[u]:
            if dist[v] > dist[u] + w:
                dist[v] = dist[u] + w
                heapq.heappush(heap, (dist[v], v))

    return dist
```

建图：

```python
n, m = 5, 6
graph = [[] for _ in range(n + 1)]

edges = [
    (1, 2, 3),
    (1, 3, 5),
    (2, 3, 1),
]

for u, v, w in edges:
    graph[u].append((v, w))
    graph[v].append((u, w))

dist = dijkstra(n, graph, 1)
```

## 2. Floyd

适合点数小的多源最短路。

```python
INF = 10 ** 18

n = 5
dist = [[INF] * (n + 1) for _ in range(n + 1)]

for i in range(1, n + 1):
    dist[i][i] = 0

edges = [
    (1, 2, 3),
    (2, 3, 4),
]

for u, v, w in edges:
    dist[u][v] = min(dist[u][v], w)
    dist[v][u] = min(dist[v][u], w)

for k in range(1, n + 1):
    for i in range(1, n + 1):
        for j in range(1, n + 1):
            dist[i][j] = min(dist[i][j], dist[i][k] + dist[k][j])
```

复杂度：

```text
O(n^3)
```

一般 `n <= 400` 才考虑。

# 十四、拓扑排序模板

适合有向无环图。

```python
from collections import deque

def topo_sort(n, graph, indeg):
    q = deque()

    for i in range(1, n + 1):
        if indeg[i] == 0:
            q.append(i)

    order = []

    while q:
        u = q.popleft()
        order.append(u)

        for v in graph[u]:
            indeg[v] -= 1

            if indeg[v] == 0:
                q.append(v)

    if len(order) == n:
        return order
    else:
        return None
```

# 十五、最小生成树 Kruskal

```python
class DSU:
    def __init__(self, n):
        self.parent = list(range(n + 1))

    def find(self, x):
        while self.parent[x] != x:
            self.parent[x] = self.parent[self.parent[x]]
            x = self.parent[x]
        return x

    def union(self, a, b):
        ra = self.find(a)
        rb = self.find(b)

        if ra == rb:
            return False

        self.parent[rb] = ra
        return True


def kruskal(n, edges):
    edges.sort(key=lambda x: x[2])

    dsu = DSU(n)
    total = 0
    cnt = 0

    for u, v, w in edges:
        if dsu.union(u, v):
            total += w
            cnt += 1

            if cnt == n - 1:
                break

    if cnt == n - 1:
        return total
    else:
        return -1
```

```python
def qpow(a, b, mod):
    res = 1

    while b:
        if b & 1:
            res = res * a % mod

        a = a * a % mod
        b >>= 1

    return res
```

## 4. 埃氏筛

```python
def get_primes(n):
    is_prime = [True] * (n + 1)
    is_prime[0] = False
    is_prime[1] = False

    primes = []

    for i in range(2, n + 1):
        if is_prime[i]:
            primes.append(i)

            j = i * i
            while j <= n:
                is_prime[j] = False
                j += i

    return primes, is_prime
```

## 5. 分解质因数

```python
def factorize(x):
    res = []

    d = 2
    while d * d <= x:
        if x % d == 0:
            cnt = 0

            while x % d == 0:
                x //= d
                cnt += 1

            res.append((d, cnt))

        d += 1

    if x > 1:
        res.append((x, 1))

    return res
```

# 十九、递归与记忆化搜索

## 1. lru_cache

```python
from functools import lru_cache

@lru_cache(None)
def dfs(x):
    if x <= 1:
        return 1

    return dfs(x - 1) + dfs(x - 2)

print(dfs(10))
```

# 二十三、做题时的判断思路

看到题目后先看数据范围。

## 1. 数据范围和复杂度大致对应

```text
n <= 20        可以考虑状态压缩、子集枚举、DFS
n <= 100       可以考虑 O(n^3)
n <= 1000      可以考虑 O(n^2)
n <= 10^5      一般需要 O(n log n) 或 O(n)
n <= 10^6      一般需要 O(n)
```

---

## 2. 常见题型对应算法

```text
区间求和              前缀和
区间修改              差分
矩阵区域求和          二维前缀和
最短步数              BFS
连通块数量            DFS / BFS / 并查集
最短路径              Dijkstra / Floyd
排序后找答案          排序 / 贪心 / 双指针
最大最小的最优值      二分答案
方案数                DP / 组合数
选或不选              01 背包 / 状态压缩
无限选择              完全背包
日期问题              datetime / 手写日期
字符串统计            Counter / 字典
```

# 二十四、蓝桥杯 Python 注意点

## 1. 优先使用 `sys.stdin.read()`

```python
data = sys.stdin.read().split()
```

比多次 `input()` 更适合大输入。

---

## 2. 输出多行时用 join

```python
print('\n'.join(ans))
```

---

## 3. 避免无意义递归

Python 递归容易爆栈。能用 BFS 或循环就优先用循环。

---

## 4. 注意整数不会溢出

Python 整数无限大，但大数运算会慢。能取模就及时取模。

---

## 5. 注意下标

蓝桥杯题目经常给的是：

```text
1 到 n
```

代码里可以开：

```python
a = [0] + list(map(int, data[1:1 + n]))
```

这样方便使用 1 下标。

---

## 6. 注意样例可能不完整

有些练习平台复制题面时，样例输入行数可能和 `n` 不一致。正式提交时以题目真实输入为准。
