
## 1. 深度优先搜索 (DFS) & 广度优先搜索 (BFS)

### 1.1 DFS 模板 (递归回溯)

**常考题型**：排列组合、N 皇后问题、迷宫求所有路径、树的遍历。

**特征**：执着地走到黑，撞南墙才回头。需要重点关注“恢复现场（回溯）”。

C++

```
bool st[N]; // 记录每个点是否被访问过

void dfs(int u) {
    if (到达终点/满足边界条件) {
        // 记录答案
        return;
    }
    
    // 遍历当前节点的所有选择
    for (int i = 0; i < n; i ++ ) {
        if (!st[i]) {
            st[i] = true; // 保护现场
            // 记录当前选择
            dfs(i + 1); // 递归到下一层
            st[i] = false; // 恢复现场 (回溯)
        }
    }
}
```

### 1.2 BFS 模板 (队列)

**常考题型**：无权图/边权均为 1 的最短路、迷宫最短步数、层序遍历、拓扑排序。

**特征**：像波纹一样向外扩散，第一次搜到目标时就是最短路径。

C++

```
int d[N];    // 存储起始点到其他点的最短距离
int q[N];    // 队列
bool st[N];  // 判重数组 (在某些不需要回头的BFS中，直接用 d[i] != -1 判重也可)

int bfs() {
    int hh = 0, tt = 0;
    q[0] = 1; // 起点入队
    memset(d, -1, sizeof d); // 初始化距离
    d[1] = 0;

    while (hh <= tt) {
        int t = q[hh ++ ]; // 取出队头
        if (t == 终点) return d[t];
        
        for (遍历 t 的所有相邻节点 j) {
            if (d[j] == -1) { // 没走过
                d[j] = d[t] + 1;
                q[ ++ tt] = j; // 压入队尾
            }
        }
    }
    return -1;
}
```

## 2. 图的存储

**极度重要**：后续所有最短路、最小生成树算法，只要用到图，90% 都是用这个**邻接表**模板存储。

C++

```
// h[N] 是各个节点的链表头，e[M] 是终点，w[M] 是边权，ne[M] 是下一个节点的指针
// N 是点数，M 是边数 (无向图 M 要开 N 的 2 倍)
int h[N], e[M], w[M], ne[M], idx;

void init() {
    memset(h, -1, sizeof h);
    idx = 0;
}

// 添加一条从 a 到 b，权重为 c 的有向边
void add(int a, int b, int c) {
    e[idx] = b, w[idx] = c, ne[idx] = h[a], h[a] = idx ++ ;
}
```

## 3. 单源最短路 (Shortest Path)

_面试高频，务必熟练默写 Dijkstra。_

### 3.1 朴素 Dijkstra (稠密图: 边多点少)

**时间复杂度**：$O(N^2)$。适用于点数 $N \le 500$ 的稠密图。

C++

```
int g[N][N];  // 稠密图用邻接矩阵存图
int dist[N];  // 距离数组
bool st[N];   // 标记是否已经确定最短路

int dijkstra() {
    memset(dist, 0x3f, sizeof dist); // 0x3f3f3f3f 是无穷大的神仙常量
    dist[1] = 0;

    for (int i = 0; i < n - 1; i ++ ) {
        int t = -1; // 在还未确定最短路的点中，寻找距离最小的点
        for (int j = 1; j <= n; j ++ )
            if (!st[j] && (t == -1 || dist[t] > dist[j]))
                t = j;

        // 用 t 更新其他点的距离
        for (int j = 1; j <= n; j ++ )
            dist[j] = min(dist[j], dist[t] + g[t][j]);

        st[t] = true;
    }
    if (dist[n] == 0x3f3f3f3f) return -1;
    return dist[n];
}
```

### 3.2 堆优化 Dijkstra (稀疏图: 点多边少)

**时间复杂度**：$O(M \log N)$。适用于点数 $N \sim 10^5$ 的稀疏图。

**核心**：用 `priority_queue` 替代上面找最小距离点 $O(N)$ 的循环。

C++

```
typedef pair<int, int> PII; // {距离, 节点编号}
int dist[N];
bool st[N];

int dijkstra() {
    memset(dist, 0x3f, sizeof dist);
    dist[1] = 0;
    priority_queue<PII, vector<PII>, greater<PII>> heap; // 小根堆
    heap.push({0, 1}); 

    while (heap.size()) {
        auto t = heap.top();
        heap.pop();

        int ver = t.second, distance = t.first;
        if (st[ver]) continue; // 冗余备份，出队过的点不再处理
        st[ver] = true;

        for (int i = h[ver]; i != -1; i = ne[i]) {
            int j = e[i];
            if (dist[j] > distance + w[i]) {
                dist[j] = distance + w[i];
                heap.push({dist[j], j});
            }
        }
    }
    if (dist[n] == 0x3f3f3f3f) return -1;
    return dist[n];
}
```

### 3.3 SPFA (存在负权边的图)

**特点**：队列优化的 Bellman-Ford，通常 $O(M)$，最坏被卡到 $O(NM)$。不仅能求带负权边的最短路，还能**判负环**。

C++

```
int dist[N];
bool st[N]; // 记录点是否在队列中

int spfa() {
    memset(dist, 0x3f, sizeof dist);
    dist[1] = 0;

    int q[N], hh = 0, tt = -1; // 队列 (也可以用 queue<int>)
    q[ ++ tt] = 1;
    st[1] = true;

    while (hh <= tt) {
        int t = q[hh ++ ];
        if (hh == N) hh = 0; // 循环队列防溢出
        st[t] = false; // 出队后取消标记

        for (int i = h[t]; i != -1; i = ne[i]) {
            int j = e[i];
            if (dist[j] > dist[t] + w[i]) {
                dist[j] = dist[t] + w[i];
                if (!st[j]) { // 如果不在队列中，才入队
                    q[ ++ tt] = j;
                    if (tt == N) tt = 0;
                    st[j] = true;
                }
            }
        }
    }
    if (dist[n] == 0x3f3f3f3f) return -1;
    return dist[n];
}
```

## 4. 多源最短路

### 4.1 Floyd 算法

**时间复杂度**：$O(N^3)$。可以求**任意两个点**之间的最短路。代码极短，必须背会。

C++

```
int d[N][N]; // d[i][j] 初始化为 i 到 j 的直接边权，没有边就是 INF，d[i][i] = 0

void floyd() {
    // k 必须在最外层！
    for (int k = 1; k <= n; k ++ )
        for (int i = 1; i <= n; i ++ )
            for (int j = 1; j <= n; j ++ )
                d[i][j] = min(d[i][j], d[i][k] + d[k][j]);
}
```

## 5. 最小生成树 (MST)

_常用于在连通图中寻找连通所有点且总边权最小的方案（例如修桥、铺电缆网络）。_

### 5.1 Kruskal 算法

**时间复杂度**：$O(M \log M)$。最常用，结合了排序和**并查集**。

C++

```
struct Edge {
    int a, b, w;
    bool operator< (const Edge &W)const { return w < W.w; }
} edges[M]; // 只要存边就行，不需要邻接表

int p[N]; // 并查集数组
int find(int x) { if (p[x] != x) p[x] = find(p[x]); return p[x]; }

int kruskal() {
    sort(edges, edges + m); // 将边按权重从小到大排序
    for (int i = 1; i <= n; i ++ ) p[i] = i; // 初始化并查集

    int res = 0, cnt = 0; // res 存总权重，cnt 存加入的边数
    for (int i = 0; i < m; i ++ ) {
        int a = edges[i].a, b = edges[i].b, w = edges[i].w;
        a = find(a), b = find(b);
        if (a != b) { // 如果不连通
            p[a] = b; // 连通它们
            res += w;
            cnt ++ ;
        }
    }
    if (cnt < n - 1) return -1; // 树没连通，不存在生成树
    return res;
}
```

## 6. 二分图

_笔试面试偶尔会考“男女配对问题”、“课程分配问题”。_

### 6.1 染色法 (判断是否是二分图)

**思路**：相邻的点不能同色，如果出现矛盾就不是二分图。

C++

```
int color[N]; // 0 表示未染色，1 表示白色，2 表示黑色

bool dfs(int u, int c) {
    color[u] = c;
    for (int i = h[u]; i != -1; i = ne[i]) {
        int j = e[i];
        if (!color[j]) {
            if (!dfs(j, 3 - c)) return false; // 3 - 1 = 2, 3 - 2 = 1，实现交替染色
        }
        else if (color[j] == c) return false; // 颜色冲突
    }
    return true;
}

### 6.2 匈牙利算法 (求二分图最大匹配)

**特征**：渣男算法。“能不能把你的现任让给我，你去看看你的备胎？”

C++

```
int match[N]; // match[j] = a 表示右半部的妹子 j 目前和左半部的男生 a 匹配
bool st[N];   // 标记右半部的妹子在这轮寻找中是否已经被问过

bool find(int x) { // x 是左半部的男生
    for (int i = h[x]; i != -1; i = ne[i]) {
        int j = e[i]; // j 是 x 看上的妹子
        if (!st[j]) {
            st[j] = true;
            // 如果妹子 j 单身，或者妹子的现男友能找到新备胎
            if (match[j] == 0 || find(match[j])) { 
                match[j] = x; // 成功匹配
                return true;
            }
        }
    }
    return false;
}
